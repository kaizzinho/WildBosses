25   /**
26    * Simulator Battle Action Queue
27    * Pokemon Showdown - http://pokemonshowdown.com/
28    *
29    * The action queue is the core of the battle simulation. A rough overview of
30    * the core battle loop:
31    *
32    * - chosen moves/switches are added to the action queue
33    * - the action queue is sorted in speed/priority order
34    * - we go through the action queue
35    * - repeat
36    *
37    * @license MIT
38    */

40     constructor(battle) {
41       this.battle = battle;
42       this.list = [];

44       if (queueScripts)

66     /**
67      * Takes an ActionChoice, and fills it out into a full Action object.
68      *
69      * Returns an array of Actions because some ActionChoices (like mega moves)
70      * resolve to two Actions (mega evolution + use move)
71      */
72     resolveAction(action, midTurn = false) {
73       if (!action)
74         throw new Error(`Action not passed to resolveAction`);
75       if (action.choice === "pass")
76         return [];
77       const actions = [action];
78       if (!action.side && action.pokemon)
79         action.side = action.pokemon.side;
80       if (!action.move && action.moveid)
81         action.move = this.battle.dex.getActiveMove(action.moveid);
82       if (!action.order) {
83         const orders = {
84           team: 1,
85           start: 2,
86           instaswitch: 3,
87           beforeTurn: 4,
88           beforeTurnMove: 5,
89           revivalblessing: 6,
90           runUnnerve: 100,
91           runSwitch: 101,
92           runPrimal: 102,
93           switch: 103,
94           megaEvo: 104,
95           runDynamax: 105,
96           terastallize: 106,
97           priorityChargeMove: 107,
98           shift: 200,
99           // default is 200 (for moves)
100          residual: 300
101        };
102        if (action.choice in orders) {
103          action.order = orders[action.choice];
104        } else {
105          action.order = 200;
106          if (!["move", "event"].includes(action.choice)) {
107            throw new Error(`Unexpected orderless action ${action.choice}`);
108          }
109        }
110      }
111      if (!midTurn) {
112        if (action.choice === "move") {
113          if (!action.maxMove && !action.zmove && action.move.beforeTurnCallback) {
114            actions.unshift(...this.resolveAction({
115              choice: "beforeTurnMove",
116              pokemon: action.pokemon,
117              move: action.move,
118              targetLoc: action.targetLoc
119            }));
120          }
121          if (action.mega && !action.pokemon.isSkyDropped()) {
122            actions.unshift(...this.resolveAction({
123              choice: "megaEvo",
124              pokemon: action.pokemon
125            }));
126          }
127          if (action.terastallize && !action.pokemon.terastallized) {
128            actions.unshift(...this.resolveAction({
129              choice: "terastallize",
130              pokemon: action.pokemon
131            }));
132          }
133          if (action.maxMove && !action.pokemon.volatiles["dynamax"]) {
134            actions.unshift(...this.resolveAction({
135              choice: "runDynamax",
136              pokemon: action.pokemon
137            }));
138          }
139          if (!action.maxMove && !action.zmove && action.move.priorityChargeCallback) {
140            actions.unshift(...this.resolveAction({
141              choice: "priorityChargeMove",
142              pokemon: action.pokemon,
143              move: action.move
144            }));
145          }
146          action.fractionalPriority = this.battle.runEvent("FractionalPriority", action.pokemon, null, action.move, 0);
147        } else if (["switch", "instaswitch"].includes(action.choice)) {
148          if (typeof action.pokemon.switchFlag === "string") {
149            action.sourceEffect = this.battle.dex.moves.get(action.pokemon.switchFlag);
150          }
151          action.pokemon.switchFlag = false;
152        }
153      }
154      const deferPriority = this.battle.gen === 7 && action.mega && action.mega !== "done";
155      if (action.move) {
156        let target = null;
157        action.move = this.battle.dex.getActiveMove(action.move);
158        if (!action.targetLoc) {
159          target = this.battle.getRandomTarget(action.pokemon, action.move);
160          if (target)
161            action.targetLoc = action.pokemon.getLocOf(target);
162        }
163        action.originalTarget = action.pokemon.getAtLoc(action.targetLoc);
164      }
165      if (!deferPriority)
166        this.battle.getActionSpeed(action);
167      return actions;
168    }

172    prioritizeAction(action, sourceEffect) {
173      for (const [i, curAction] of this.list.entries()) {
174        if (curAction === action) {
175          this.list.splice(i, 1);
176          break;
177        }
178      }
179      action.sourceEffect = sourceEffect;
180      action.order = 3;
181      this.list.unshift(action);
182    }
183    /**
184     * Changes a pokemon's action, and inserts its new action
185     * in priority order.
186     *
187     * You'd normally want the OverrideAction event (which doesn't
188     * change priority order).
189     */
190    changeAction(pokemon, action) {
191      this.cancelAction(pokemon);
192      if (!action.pokemon)
193        action.pokemon = pokemon;
194      this.insertChoice(action);

196    addChoice(choices) {
197      if (!Array.isArray(choices))

205          }
206        }
207      }

254    /**
255     * Inserts the passed action into the action queue when it normally
256     * would have happened (sorting by priority/speed), without
257     * re-sorting the existing actions.
258     */
259    insertChoice(choices, midTurn = false) {
260      if (Array.isArray(choices)) {
261        for (const choice2 of choices) {
262          this.insertChoice(choice2);
263        }
264        return;
265      }
266      const choice = choices;
267      if (choice.pokemon) {
268        choice.pokemon.updateSpeed();
269      }
270      const actions = this.resolveAction(choice, midTurn);
271      let firstIndex = null;
272      let lastIndex = null;
273      for (const [i, curAction] of this.list.entries()) {
274        const compared = this.battle.comparePriority(actions[0], curAction);
275        if (compared <= 0 && firstIndex === null) {
276          firstIndex = i;
277        }
278        if (compared < 0) {
279          lastIndex = i;
280          break;
281        }
282      }
283      if (firstIndex === null) {
284        this.list.push(...actions);
285      } else {
286        if (lastIndex === null)
287          lastIndex = this.list.length;
288        const index = firstIndex === lastIndex ? firstIndex : this.battle.random(firstIndex, lastIndex + 1);
289        this.list.splice(index, 0, ...actions);
290      }
291    }
292    clear() {
293      this.list = [];
294    }

298      }
299      return this.list.map(

302    }
