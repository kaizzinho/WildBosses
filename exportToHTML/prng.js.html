27    *
28    * This simulates the on-cartridge PRNG used in the real games.
29    *
30    * In addition to potentially allowing us to read replays from in-game,
31    * this also makes it possible to record an "input log" (a seed +
32    * initial teams + move/switch decisions) and "replay" a simulation to
33    * get the same result.
34    *
35    * @license MIT license
36    */
37   class PRNG {
38     /** Creates a new source of randomness for the given seed. */
39     constructor(seed = null) {
40       if (!seed)
41         seed = PRNG.generateSeed();
42       this.initialSeed = seed.slice();
43       this.seed = seed.slice();
44     }
45     /**
46      * Getter to the initial seed.
47      *
48      * This should be considered a hack and is only here for backwards compatibility.
49      */
50     get startingSeed() {
51       return this.initialSeed;
52     }
53     /**
54      * Creates a clone of the current PRNG.
55      *
56      * The new PRNG will have its initial seed set to the seed of the current instance.
57      */

59       return new PRNG(this.seed);
60     }
61     /**
62      * Retrieves the next random number in the sequence.
63      * This function has three different results, depending on arguments:
64      * - random() returns a real number in [0, 1), just like Math.random()
65      * - random(n) returns an integer in [0, n)
66      * - random(m, n) returns an integer in [m, n)
67      * m and n are converted to integers via Math.floor. If the result is NaN, they are ignored.
68      */
69     next(from, to) {
70       this.seed = this.nextFrame(this.seed);
71       let result = (this.seed[0] << 16 >>> 0) + this.seed[1];
72       if (from)
73         from = Math.floor(from);
74       if (to)
75         to = Math.floor(to);
76       if (from === void 0) {
77         result = result / 4294967296;
78       } else if (!to) {
79         result = Math.floor(result * from / 4294967296);
80       } else {
81         result = Math.floor(result * (to - from) / 4294967296) + from;
82       }
83       return result;
84     }
85     /**
86      * Flip a coin (two-sided die), returning true or false.
87      *
88      * This function returns true with probability `P`, where `P = numerator
89      * / denominator`. This function returns false with probability `1 - P`.
90      *
91      * The numerator must be a non-negative integer (`>= 0`).
92      *
93      * The denominator must be a positive integer (`> 0`).
94      */

96       return this.next(denominator) < numerator;
97     }
98     /**
99      * Return a random item from the given array.
100     *
101     * This function chooses items in the array with equal probability.
102     *
103     * If there are duplicate items in the array, each duplicate is
104     * considered separately. For example, sample(['x', 'x', 'y']) returns
105     * 'x' 67% of the time and 'y' 33% of the time.
106     *
107     * The array must contain at least one item.
108     *
109     * The array must not be sparse.
110     */
111    sample(items) {
112      if (items.length === 0) {
113        throw new RangeError(`Cannot sample an empty array`);
114      }
115      const index = this.next(items.length);
116      const item = items[index];
117      if (item === void 0 && !Object.prototype.hasOwnProperty.call(items, index)) {
118        throw new RangeError(`Cannot sample a sparse array`);
119      }
120      return item;
121    }
122    /**
123     * A Fisher-Yates shuffle. This is how the game resolves speed ties.
124     *
125     * At least according to V4 in
126     * https://github.com/smogon/pokemon-showdown/issues/1157#issuecomment-214454873
127     */
128    shuffle(items, start = 0, end = items.length) {
129      while (start < end - 1) {
130        const nextIndex = this.next(start, end);
131        if (start !== nextIndex) {
132          [items[start], items[nextIndex]] = [items[nextIndex], items[start]];
133        }
134        start++;
135      }
136    }
137    /**
138     * Calculates `a * b + c` (with 64-bit 2's complement integers)
139     *
140     * If you've done long multiplication, this is the same thing.
141     */
142    multiplyAdd(a, b, c) {
143      const out = [0, 0, 0, 0];
144      let carry = 0;
145      for (let outIndex = 3; outIndex >= 0; outIndex--) {
146        for (let bIndex = outIndex; bIndex < 4; bIndex++) {
147          const aIndex = 3 - (bIndex - outIndex);
148          carry += a[aIndex] * b[bIndex];
149        }
150        carry += c[outIndex];
151        out[outIndex] = carry & 65535;
152        carry >>>= 16;
153      }
154      return out;
155    }
156    /**
157     * The RNG is a Linear Congruential Generator (LCG) in the form: `x_{n + 1} = (a x_n + c) % m`
158     *
159     * Where: `x_0` is the seed, `x_n` is the random number after n iterations,
160     *
161     * ````
162     * a = 0x5D588B656C078965
163     * c = 0x00269EC3
164     * m = 2^64
165     * ````
166     */
167    nextFrame(seed, framesToAdvance = 1) {
168      const a = [23896, 35685, 27655, 35173];
169      const c = [0, 0, 38, 40643];
170      for (let i = 0; i < framesToAdvance; i++) {
171        seed = this.multiplyAdd(seed, a, c);
172      }
173      return seed;
174    }
175    static generateSeed() {
176      return [

178        Math.floor(Math.random() * 65536),
179        Math.floor(Math.random() * 65536),
180        Math.floor(Math.random() * 65536)
181      ];
182    }
